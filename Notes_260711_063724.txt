<script>
let allStrains = [];

async function loadStrains(category = 'all') {
const res = await fetch(/api/strains?category=${category}`);
allStrains = await res.json();
renderStrains(allStrains);
}

function renderStrains(strains) {
const list = document.getElementById('strainList');
list.innerHTML = strains.map(s => &lt;div class="strain-item"&gt; &lt;div&gt; &lt;span class="strain-name"&gt;${s.name}</span>
<span class="strain-type">[{s.year}</span>
<span style="font-size:0.7rem;color:#666;">${s.capabilities.join(', ')}&lt;/span&gt; &lt;/div&gt; &lt;/div&gt;).join('');
}

function filterStrains(category) {
loadStrains(category);
}

function searchStrains() {
const q = document.getElementById('searchInput').value.toLowerCase();
const filtered = allStrains.filter(s =>
s.name.toLowerCase().includes(q) ||
s.description.toLowerCase().includes(q)
);
renderStrains(filtered);
}

async function loadStats() {
const res = await fetch('/api/stats');
const stats = await res.json();
const container = document.getElementById('stats-container');
container.innerHTML = &lt;div style="font-size:0.8rem;margin-top:1rem;"&gt;${Object.entries(stats.by_type).map(([k,v]) => <div&gt;${k}: ${v}&lt;/div>).join('')}
</div>
`;
}

async function executeCommand() {
const input = document.getElementById('cmdInput');
const terminal = document.getElementById('terminal');
const cmd = input.value;
if (!cmd) return;

terminal.innerHTML += \n$${cmd}\n;
input.value = '';

try {
const res = await fetch('/api/exec', {
method: 'POST',
headers: {'Content-Type': 'application/json'},
body: JSON.stringify({command: cmd})
});
const data = await res.json();
if (data.stdout) terminal.innerHTML += data.stdout;
if (data.stderr) terminal.innerHTML += <span style="color:#ff0000;"&gt;${data.stderr}</span>; if (data.error) terminal.innerHTML +=<span style="color:#ff0000;">ERROR: ${data.error}&lt;/span>;
} catch(e) {
terminal.innerHTML += <span style="color:#ff0000;"&gt;FAILED: ${e.message}&lt;/span>;
}
terminal.scrollTop = terminal.scrollHeight;
}

loadStrains();
loadStats();
</script>
