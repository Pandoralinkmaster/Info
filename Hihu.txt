class SessionManager:
@staticmethod
def create():
t = secrets.token_urlsafe(32)
SESSIONS[t] = {"created": time.time(), "last_used": time.time(), "requests": 0}
return t
@staticmethod
def validate(t):
if t not in SESSIONS:
return False
s = SESSIONS[t]
if time.time() - s["created"] > SESSION_TIMEOUT:
del SESSIONS[t]
return False
s["last_used"] = time.time()
s["requests"] += 1
return True
@staticmethod
def cleanup():
now = time.time()
for t in [k for k, v in SESSIONS.items() if now - v["created"] > SESSION_TIMEOUT]:
del SESSIONS[t]

class RateLimiter:
@staticmethod
def check(ip):
now = time.time()
e = RATE_LIMIT[ip]
if now - e["window_start"] > RATE_LIMIT_WINDOW:
e["count"] = 1
e["window_start"] = now
return True

if e["count"] > MAX_REQUESTS:

class Block:
def init(self, idx, ts, dna, prev):
self.index = idx
self.timestamp = ts
self.dna_data = dna
self.previous_hash = prev
self.nonce = 0
self.hash = self.calc()
def calc(self):
return hashlib.sha3_256(f"{self.index}{self.timestamp}{self.dna_data}{self.previous_hash}{self.nonce}".encode()).hexdigest()
def mine(self, diff=4):
target = "0" * diff
while self.hash[:diff] != target:
self.nonce += 1
self.hash = self.calc()
logger.info("Block " + str(self.index) + " mined: " + self.hash[:16] + "...")

class QuantumCrypto:

r = data.encode() if isinstance(data, str) else data
for _ in range(rounds):
r = hashlib.sha3_512(r).digest()
r = hashlib.blake2b(r).digest()
return hashlib.sha3_512(r).hexdigest()

class Block:
def init(self, idx, ts, dna, prev):

self.timestamp = ts
self.dna_data = dna

while self.hash[:diff] != target:
self.nonce += 1
self.hash = self.calc()

const canvas = document.getElementById('canvas');
const ctx = canvas.getContext('2d');
let w, h, particles = [];
function resize() { w = canvas.width = window.innerWidth; h = canvas.height = window.innerHeight; }
window.addEventListener('resize', resize); resize();

class P {
constructor() { this.x=Math.random()*w; this.y=Math.random()*h; this.vx=(Math.random()-0.5)*0.5; this.vy=(Math.random()-0.5)0.5; this.s=Math.random()2+0.5; this.a=Math.random()0.5+0.2; this.c=Math.random()>0.85?'#ff0':(Math.random()>0.9?'#f0f':'#f00'); }
update() { this.x+=this.vx; this.y+=this.vy; if(this.x<0||this.x>w)this.vx=-1; if(this.y<0||this.y>h)this.vy=-1; }
draw() { ctx.beginPath(); ctx.arc(this.x,this.y,this.s,0,Math.PI2); ctx.fillStyle=this.c==='#f00'?rgba(255,0,0,${this.a}):(this.c==='#ff0'?rgba(255,255,0,{this.a})`); ctx.fill(); }
}
for(let i=0;i<400;i++) particles.push(new P());

function drawCon() {
for(let i=0;i<particles.length;i++){
for(let j=i+1;j<particles.length;j++){
const dx=particles[i].x-particles[j].x, dy=particles[i].y-particles[j].y, dist=Math.sqrt(dxdx+dydy);
if(dist<120){ ctx.beginPath(); ctx.moveTo(particles[i].x,particles[i].y); ctx.lineTo(particles[j].x,particles[j].y); ctx.strokeStyle=rgba(255,0,0,${0.1*(1-dist/120)}); ctx.stroke(); }
}
}
}
function anim() { ctx.fillStyle='rgba(0,0,0,0.05)'; ctx.fillRect(0,0,w,h); particles.forEach(p=>{p.update();p.draw();}); drawCon(); requestAnimationFrame(anim); }
anim();
